@echo off
if exist "Locker" goto UNLOCK
if not exist "Private" goto MDLOCKER
:CONFIRM
echo Are you sure to lock the folder? (Y/N)
set/p "cho=>"
if %cho%==Y goto LOCK
if %cho%==y goto LOCK
if %cho%==N goto END
if %cho%==n goto END
:LOCK
ren Private "Locker"
attrib +h +s "Locker"
echo Folder locked
goto End
:UNLOCK
echo Enter password to unlock:
set/p "pass=>"
if NOT %pass%==YourPasswordHere goto FAIL
attrib -h -s "Locker"
ren "Locker" Private
echo Folder unlocked
goto End
:FAIL
echo Wrong password
goto end
:MDLOCKER
md Private
echo Private folder created
goto End
:End
